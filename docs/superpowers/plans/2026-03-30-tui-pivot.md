# Talat TUI Pivot Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Pivot Talat from a Python/React web stack to a self-contained C++ binary with a terminal UI, a realistic 3-trader market model anchored to a shared GBM latent price, and an expanded benchmark harness.

**Architecture:** The C++ core (`OrderBook`, `OrderMatcher`, `EventBus`, `Exchange`, `Portfolio`) stays unchanged. A new `LatentPrice` component drives three redesigned traders (`MarketMaker`, `InformedTrader`, `NoiseTrader`) through a rebuilt `TraderRegistry`. A `StatsTracker` subscribes to `EventBus` for live latency/throughput metrics. An ftxui-based `TUI` class renders everything in a three-panel terminal layout with a command bar.

**Tech Stack:** C++20, CMake FetchContent (ftxui 5.0.0), Boost (existing), Google Test (existing), Google Benchmark (existing)

Create `include/latent_price.h` (header-only):

```cpp
#pragma once
#include <atomic>
#include <cmath>
#include <random>
#include "order.h"

class LatentPrice {
public:
    explicit LatentPrice(Price seed_price, double sigma = 0.0003)
        : sigma_(sigma)
        , log_price_(std::log(static_cast<double>(seed_price) / 10000.0))
        , rng_(std::random_device{}())
        , dist_(0.0, 1.0)
    {
        price_.store(seed_price, std::memory_order_release);
    }

    // Advance GBM by one step. Call from tick thread only (not thread-safe for
    // concurrent tick() calls, but get() is safe from any thread).
    void tick() {
        double z = dist_(rng_);
        log_price_ += sigma_ * z;
        Price new_price = static_cast<Price>(std::exp(log_price_) * 10000.0);
        // Guard against underflow to zero (extremely rare with normal sigma)
        if (new_price == 0) new_price = 1;
        price_.store(new_price, std::memory_order_release);
    }

    // Thread-safe read from any thread.
    Price get() const {
        return price_.load(std::memory_order_acquire);
    }

    void set_sigma(double sigma) { sigma_ = sigma; }
    double sigma() const { return sigma_; }

private:
    std::atomic<Price> price_{};
    double log_price_;
    double sigma_;
    std::mt19937 rng_;
    std::normal_distribution<double> dist_;
};
```

```cpp
#pragma once
#include <vector>
#include "trader.h"
#include "latent_price.h"

class MarketMaker : public Trader {
public:
    MarketMaker(TraderId id, std::string name, uint64_t balance,
                const LatentPrice& latent,
                Price half_spread = 20000);  // $2.00 default

    void tick(Price last_price, SubmitFn submit, CancelFn cancel) override;
    void on_fill(const Fill& fill) override;

    void set_half_spread(Price spread) { half_spread_ = spread; }
    Price half_spread() const { return half_spread_; }

private:
    const LatentPrice& latent_;
    Price half_spread_;

    std::vector<OrderId> resting_bids_;
    std::vector<OrderId> resting_asks_;

    // Adverse selection window
    uint64_t fills_in_window_{0};
    uint64_t ticks_in_window_{0};
    static constexpr uint64_t WINDOW_TICKS = 20;
    static constexpr double MAX_SPREAD_MULTIPLIER = 3.0;
};
```

- [ ] **Step 4: Write market_maker.cpp**

Create `src/traders/market_maker.cpp`:

```cpp
#include "traders/market_maker.h"
#include <algorithm>
#include <cmath>

MarketMaker::MarketMaker(TraderId id, std::string name, uint64_t balance,
                         const LatentPrice& latent, Price half_spread)
    : Trader(id, std::move(name), balance)
    , latent_(latent)
    , half_spread_(half_spread)
{}

void MarketMaker::on_fill(const Fill& fill) {
    Trader::on_fill(fill);
    if (fill.maker_trader_id == id_) {
        fills_in_window_++;
    }
}

void MarketMaker::tick(Price last_price, SubmitFn submit, CancelFn cancel) {
    // Cancel all resting orders from the previous tick
    for (OrderId id : resting_bids_) cancel(id);
    for (OrderId id : resting_asks_) cancel(id);
    resting_bids_.clear();
    resting_asks_.clear();

    // Compute effective spread (widen under adverse selection)
    ticks_in_window_++;
    Price effective_spread = half_spread_;
    if (ticks_in_window_ >= WINDOW_TICKS) {
        double fill_rate = static_cast<double>(fills_in_window_) / ticks_in_window_;
        if (fill_rate > 0.3) {
            double multiplier = std::min(1.0 + fill_rate * 2.0, MAX_SPREAD_MULTIPLIER);
            effective_spread = static_cast<Price>(half_spread_ * multiplier);
        }
        fills_in_window_ = 0;
        ticks_in_window_ = 0;
    }

    Price fair_value = latent_.get();
    if (fair_value == 0) return;

    Price bid_price = fair_value > effective_spread ? fair_value - effective_spread : 1;
    Price ask_price = fair_value + effective_spread;

    OrderId bid_id = next_id();
    submit(LimitOrder{bid_id, id_, Side::Buy, bid_price, 10, TimeInForce::GTC, {}});
    resting_bids_.push_back(bid_id);

    OrderId ask_id = next_id();
    submit(LimitOrder{ask_id, id_, Side::Sell, ask_price, 10, TimeInForce::GTC, {}});
    resting_asks_.push_back(ask_id);
}
```

```cpp
#include "traders/informed_trader.h"
#include <cmath>

InformedTrader::InformedTrader(TraderId id, std::string name, uint64_t balance,
                               const LatentPrice& latent,
                               double signal_sigma, double entry_threshold)
    : Trader(id, std::move(name), balance)
    , latent_(latent)
    , signal_sigma_(signal_sigma)
    , entry_threshold_(entry_threshold)
    , rng_(std::random_device{}())
    , noise_(0.0, signal_sigma > 0.0 ? signal_sigma : 1e-10)
{}

void InformedTrader::tick(Price last_price, SubmitFn submit, CancelFn cancel) {
    tick_count_++;
    if (tick_count_ % SIGNAL_INTERVAL != 0) return;
    if (last_price == 0) return;

    Price fair_value = latent_.get();
    if (fair_value == 0) return;

    // Noisy signal around true fair value
    double epsilon = noise_(rng_);
    double signal_price = fair_value * (1.0 + epsilon);

    double deviation = (signal_price - static_cast<double>(last_price))
                       / static_cast<double>(last_price);

    if (std::abs(deviation) < entry_threshold_) return;

    Side side = deviation > 0.0 ? Side::Buy : Side::Sell;

    // Aggressive limit: slightly past signal to ensure execution
    double price_f = side == Side::Buy
        ? signal_price * 1.001
        : signal_price * 0.999;

    Price order_price = static_cast<Price>(price_f);
    if (order_price == 0) order_price = 1;

    submit(LimitOrder{next_id(), id_, side, order_price, 5, TimeInForce::IOC, {}});
}
```

```cpp
#include "traders/noise_trader.h"
#include <algorithm>
#include <cmath>

NoiseTrader::NoiseTrader(TraderId id, std::string name, uint64_t balance, double lambda)
    : Trader(id, std::move(name), balance)
    , rng_(std::random_device{}())
    , arrivals_(std::max(lambda, 0.0))
{}

void NoiseTrader::tick(Price last_price, SubmitFn submit, CancelFn /*cancel*/) {
    if (last_price == 0) return;

    int num_orders = arrivals_(rng_);
    for (int i = 0; i < num_orders; i++) {
        Side side = side_dist_(rng_) == 0 ? Side::Buy : Side::Sell;
        Quantity qty = std::max(Quantity{1}, static_cast<Quantity>(size_dist_(rng_)));

        if (type_dist_(rng_) < 0.6) {
            // Limit order: price within ~10bps of touch
            double bps_offset = offset_dist_(rng_) * 5.0;  // up to ±5bps extra
            double price_f;
            if (side == Side::Buy) {
                price_f = last_price * (1.0 - 0.0010 + bps_offset * 0.0001);
            } else {
                price_f = last_price * (1.0 + 0.0010 + bps_offset * 0.0001);
            }
            Price price = static_cast<Price>(std::max(price_f, 1.0));
            submit(LimitOrder{next_id(), id_, side, price, qty, TimeInForce::GTC, {}});
        } else {
            // Market order
            submit(MarketOrder{next_id(), id_, side, qty, TimeInForce::GTC, {}});
        }
    }
}
```

```cpp
    : matcher_(matcher)
    , bus_(bus)
    , latent_(seed_price, sigma)
{}

TraderRegistry::~TraderRegistry() {
    stop();
    for (auto token : fill_tokens_) bus_.unsubscribe(token);
}

void TraderRegistry::start() {
    subscribe_to_fills();
    running_.store(true);
    tick_thread_ = std::thread(&TraderRegistry::tick_loop, this);
}

void TraderRegistry::stop() {
    running_.store(false);
    if (tick_thread_.joinable()) tick_thread_.join();
}

void TraderRegistry::subscribe_to_fills() {
    fill_tokens_.push_back(bus_.subscribe<FillEvent>([this](const FillEvent& e) {
        std::unique_lock lock(mutex_);
        for (auto& [id, entry] : traders_) {
            entry.trader->on_fill(e.fill);
        }
    }));
}

void TraderRegistry::tick_loop() {
    while (running_.load()) {
        latent_.tick();

        if (!paused_.load()) {
            Price last_price = latent_.get();

            auto submit_fn = [this](Order order) {
                matcher_.submit(std::move(order));
            };
            auto cancel_fn = [this](OrderId id) -> bool {
                return matcher_.cancel(id);
            };

            std::unique_lock lock(mutex_);
            for (auto& [id, entry] : traders_) {
                if (entry.active) {
                    entry.trader->tick(last_price, submit_fn, cancel_fn);
                }
            }
        }

        std::this_thread::sleep_for(
            std::chrono::milliseconds(tick_interval_ms_.load()));
    }
}
```

```cpp
            auto now = Clock::now();
            auto latency_us = std::chrono::duration_cast<std::chrono::microseconds>(
                now - e.submit_tp).count();
            if (latency_us >= 0) {
                std::lock_guard lock(mutex_);
                prune(now);
                latencies_.push_back({now, static_cast<uint64_t>(latency_us)});
                order_times_.push_back(now);
            }
        }));
    }

    ~StatsTracker() {
        // Note: bus must outlive StatsTracker. Caller must ensure unsubscription
        // if StatsTracker is destroyed before the bus.
    }

    Snapshot snapshot() const {
        std::lock_guard lock(mutex_);
        auto now = Clock::now();

        // Throughput: orders in the last 5 seconds
        size_t recent_orders = 0;
        for (const auto& tp : order_times_) {
            if (now - tp <= WINDOW) recent_orders++;
        }
        double ops = static_cast<double>(recent_orders) /
                     std::chrono::duration<double>(WINDOW).count();

        if (latencies_.empty()) return {0, 0, ops};

        // Percentiles over window
        std::vector<uint64_t> vals;
        vals.reserve(latencies_.size());
        for (const auto& r : latencies_) {
            if (now - r.ts <= WINDOW) vals.push_back(r.latency_us);
        }
        if (vals.empty()) return {0, 0, ops};

        std::sort(vals.begin(), vals.end());
        uint64_t p50 = vals[vals.size() * 50 / 100];
        uint64_t p99 = vals[vals.size() * 99 / 100];

        return {p50, p99, ops};
    }

private:
    using Clock = std::chrono::steady_clock;
    static constexpr auto WINDOW = std::chrono::seconds(5);

    struct LatencyRecord { Clock::time_point ts; uint64_t latency_us; };

    void prune(const Clock::time_point& now) {
        while (!latencies_.empty() && now - latencies_.front().ts > WINDOW)
            latencies_.pop_front();
        while (!order_times_.empty() && now - order_times_.front() > WINDOW)
            order_times_.pop_front();
    }

    mutable std::mutex mutex_;
    std::deque<LatencyRecord> latencies_;
    std::deque<Clock::time_point> order_times_;
    std::vector<SubscriptionToken> tokens_;
};
```

```cpp
void TUI::run() {
    auto screen = ScreenInteractive::Fullscreen();

    // Post a screen refresh whenever the book updates
    tokens_.push_back(exchange_.event_bus().subscribe<BookUpdateEvent>(
        [&screen](const BookUpdateEvent&) {
            screen.PostEvent(Event::Custom);
        }));

    auto input_component = Input(&command_input_, "> ");
    auto component = CatchEvent(input_component, [&](Event event) -> bool {
        if (event == Event::Return) {
            handle_command(command_input_);
            command_input_.clear();
            return true;
        }
        if (event == Event::Escape) {
            command_input_.clear();
            return true;
        }
        return false;
    });

    auto renderer = Renderer(component, [&]() -> Element {
        return vbox({
            render_header(),
            hbox({
                render_book(),
                render_fills(),
                render_stats(),
            }) | flex,
            hbox({text(" > "), input_component->Render()}) | border,
        });
    });

    screen.Loop(renderer);
}
```

```cpp
void TUI::handle_order_command(const std::string& cmd) {
    // Supported formats:
    //   buy <qty> @ <price>   → limit buy
    //   sell <qty> @ <price>  → limit sell
    //   buy <qty>             → market buy
    //   sell <qty>            → market sell
    //   cancel <order_id>     → cancel resting order
    //   q or quit             → exit (handled in run() via Escape/quit key)

    std::istringstream ss(cmd);
    std::string token;
    ss >> token;

    auto set_status = [this](const std::string& msg) {
        std::lock_guard lock(state_mutex_);
        status_message_ = msg;
    };

    if (token == "cancel") {
        OrderId id = 0;
        if (!(ss >> id)) { set_status("Usage: cancel <order_id>"); return; }
        bool ok = exchange_.cancel_order(id);
        set_status(ok ? "Cancelled #" + std::to_string(id)
                      : "Cancel failed (order not found or already filled)");
        return;
    }

    Side side;
    if (token == "buy")       side = Side::Buy;
    else if (token == "sell") side = Side::Sell;
    else { set_status("Unknown command. Type /help."); return; }

    double qty_d = 0;
    if (!(ss >> qty_d) || qty_d <= 0) {
        set_status("Usage: buy/sell <qty> [@ <price>]");
        return;
    }
    Quantity qty = static_cast<Quantity>(qty_d);
    if (qty == 0) qty = 1;

    std::string at;
    ss >> at;

    if (at == "@") {
        // Limit order
        double price_d = 0;
        if (!(ss >> price_d) || price_d <= 0) {
            set_status("Usage: buy <qty> @ <price>");
            return;
        }
```

```cpp
void TUI::handle_slash_command(const std::string& cmd) {
    std::istringstream ss(cmd.substr(1));  // strip leading /
    std::string verb;
    ss >> verb;

    auto set_status = [this](const std::string& msg) {
        std::lock_guard lock(state_mutex_);
        status_message_ = msg;
    };

    if (verb == "help") {
        set_status(
            "/vol <0.0-1.0>  /spread <$>  /speed <1-10>  "
            "/pause  /resume  /traders <mm|informed|noise> <n>");
        return;
    }

    if (verb == "vol") {
        double v = 0;
        if (!(ss >> v) || v < 0 || v > 1.0) {
            set_status("Usage: /vol <0.0–1.0>"); return;
        }
        registry_.set_sigma(v);
        set_status("Volatility set to " + std::to_string(v));
        return;
    }

    if (verb == "spread") {
        double dollars = 0;
        if (!(ss >> dollars) || dollars <= 0) {
            set_status("Usage: /spread <dollars>  e.g. /spread 2.50"); return;
        }
        Price half_spread = static_cast<Price>(dollars * 10000.0 / 2.0);
        registry_.set_market_maker_spread(half_spread);
        set_status("MM half-spread set to $" + std::to_string(dollars / 2.0));
        return;
    }

    if (verb == "speed") {
        int speed = 0;
        if (!(ss >> speed) || speed < 1 || speed > 10) {
            set_status("Usage: /speed <1–10>"); return;
        }
        // speed=1 → 2000ms interval, speed=10 → 50ms interval
        int interval_ms = 2000 / speed;
        registry_.set_tick_interval_ms(interval_ms);
        set_status("Tick interval set to " + std::to_string(interval_ms) + "ms");
        return;
    }

    if (verb == "pause") {
        registry_.pause_all();
        set_status("Market paused. Type /resume to continue.");
        return;
    }

    if (verb == "resume") {
        registry_.resume_all();
        set_status("Market resumed.");
        return;
    }

    if (verb == "traders") {
        std::string type;
        int n = 0;
        if (!(ss >> type >> n) || n < 0) {
            set_status("Usage: /traders <mm|informed|noise> <count>"); return;
        }
        if (type == "mm") {
            registry_.set_market_maker_count(static_cast<size_t>(n));
        } else if (type == "informed") {
            registry_.set_informed_count(static_cast<size_t>(n));
        } else if (type == "noise") {
            registry_.set_noise_count(static_cast<size_t>(n));
        } else {
            set_status("Unknown type. Use: mm, informed, noise"); return;
        }
        set_status("Set " + type + " count to " + std::to_string(n));
        return;
    }

    set_status("Unknown command '" + verb + "'. Type /help.");
}
```
